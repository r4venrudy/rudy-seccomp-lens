cmake_minimum_required(VERSION 3.20)
project(rudy-seccomp-lens VERSION 0.1.0 LANGUAGES CXX)

option(RSL_ENABLE_SANITIZERS "Build with AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_library(seccomp_lens_core
    src/json.cpp
    src/seccomp.cpp
)
target_include_directories(seccomp_lens_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(seccomp_lens_core PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(seccomp_lens_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(seccomp_lens_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

add_executable(seccomp-lens src/main.cpp)
target_link_libraries(seccomp-lens PRIVATE seccomp_lens_core)
if(MSVC)
    target_compile_options(seccomp-lens PRIVATE /W4 /permissive-)
else()
    target_compile_options(seccomp-lens PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

if(RSL_ENABLE_SANITIZERS AND NOT MSVC)
    target_compile_options(seccomp_lens_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(seccomp_lens_core PRIVATE -fsanitize=address,undefined)
    target_compile_options(seccomp-lens PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(seccomp-lens PRIVATE -fsanitize=address,undefined)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(seccomp-lens-tests tests/test.cpp)
    target_link_libraries(seccomp-lens-tests PRIVATE seccomp_lens_core)
    if(MSVC)
        target_compile_options(seccomp-lens-tests PRIVATE /W4 /permissive-)
    else()
        target_compile_options(seccomp-lens-tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
    if(RSL_ENABLE_SANITIZERS AND NOT MSVC)
        target_compile_options(seccomp-lens-tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(seccomp-lens-tests PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME unit COMMAND seccomp-lens-tests)
    add_test(NAME cli_lint_json COMMAND seccomp-lens lint --format json --fail-on none "${CMAKE_CURRENT_SOURCE_DIR}/examples/deny-by-default.json")
    add_test(NAME cli_probe COMMAND seccomp-lens probe "${CMAKE_CURRENT_SOURCE_DIR}/examples/deny-by-default.json" --syscall openat --args 0,0,0,0,0,0)
    add_test(NAME cli_structural_diff COMMAND seccomp-lens diff --format json "${CMAKE_CURRENT_SOURCE_DIR}/examples/deny-by-default.json" "${CMAKE_CURRENT_SOURCE_DIR}/examples/allow-by-default.json")
endif()

install(TARGETS seccomp-lens RUNTIME DESTINATION bin)
install(FILES README.md LICENSE SECURITY.md CONTRIBUTING.md DESTINATION share/doc/rudy-seccomp-lens)
install(DIRECTORY docs/ DESTINATION share/doc/rudy-seccomp-lens/docs FILES_MATCHING PATTERN "*.md")
install(DIRECTORY examples/ DESTINATION share/doc/rudy-seccomp-lens/examples)
